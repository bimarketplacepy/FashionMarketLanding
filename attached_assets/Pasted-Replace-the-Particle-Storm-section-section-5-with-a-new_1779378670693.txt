waves propagate from different points around the surface, interfering 
constructively and destructively to create standing wave patterns.

VERTEX SHADER

Compute vertex Y displacement as the sum of 5 sine waves emanating 
from different source points on the surface:

uniform float uTime;
varying float vDisplacement;
varying vec2 vUv;

void main() {
  vUv = uv;
  vec3 pos = position;
  
  // 5 wave source points distributed around the surface edges
  vec2 sources[5];
  sources[0] = vec2(-6.0, -6.0);
  sources[1] = vec2(6.0, -6.0);
  sources[2] = vec2(-6.0, 6.0);
  sources[3] = vec2(6.0, 6.0);
  sources[4] = vec2(0.0, 0.0);  // centre source for radial symmetry
  
  // 5 different frequencies (rationally related for stable patterns)
  float freqs[5];
  freqs[0] = 1.8;
  freqs[1] = 2.4;
  freqs[2] = 3.1;
  freqs[3] = 2.7;
  freqs[4] = 1.5;
  
  // Slowly modulate frequencies over time for evolving patterns
  float timeShift = uTime * 0.15;
  
  float totalDisplacement = 0.0;
  for (int i = 0; i < 5; i++) {
    vec2 sourceToPoint = pos.xy - sources[i];
    float dist = length(sourceToPoint);
    float waveValue = sin(dist * freqs[i] + timeShift * (1.0 + float(i) * 0.3));
    // Attenuate by distance for realism
    waveValue *= 1.0 / (1.0 + dist * 0.15);
    totalDisplacement += waveValue;
  }
  
  // Scale displacement to a comfortable range
  totalDisplacement *= 0.35;
  
  pos.z += totalDisplacement;
  vDisplacement = totalDisplacement;
  
  gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
}

FRAGMENT SHADER

Colour based on displacement amplitude (peaks warm, troughs cool) and 
add a subtle grid pattern to make the surface texture readable:

varying float vDisplacement;
varying vec2 vUv;
uniform float uTime;

void main() {
  // Map displacement to colour
  float t = (vDisplacement + 0.7) / 1.4;  // normalise to 0-1
  t = clamp(t, 0.0, 1.0);
  
  vec3 colTrough = vec3(0.15, 0.05, 0.4);   // deep violet
  vec3 colMid = vec3(0.05, 0.4, 0.65);      // teal-blue
  vec3 colPeak = vec3(0.95, 0.75, 0.45);    // warm gold
  
  vec3 colour;
  if (t < 0.5) {
    colour = mix(colTrough, colMid, t * 2.0);
  } else {
    colour = mix(colMid, colPeak, (t - 0.5) * 2.0);
  }
  
  // Add subtle contour lines along displacement isolines for surface 
  // definition (so the geometry reads even on flat-lit areas)
  float contour = abs(fract(vDisplacement * 4.0) - 0.5) * 2.0;
  contour = smoothstep(0.85, 1.0, contour);
  colour = mix(colour, colour * 1.4, contour * 0.3);
  
  // Glow at the peaks
  float peakGlow = smoothstep(0.7, 1.0, t);
  colour += vec3(0.4, 0.3, 0.15) * peakGlow * 0.5;
  
  gl_FragColor = vec4(colour, 1.0);
}
